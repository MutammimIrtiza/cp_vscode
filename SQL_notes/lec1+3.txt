find manager name for all employees

SELECT E.FIRST_NAME AS EMPLOYEE, M.FIRST_NAME AS MANAGER
FROM EMPLOYEES E
JOIN EMPLOYEES M
ON E.MANAGER_ID = M.EMPLOYEE_ID



##
find first name of employees, count of people getting more salary than him/her

SELECT E1.FIRST_NAME, COUNT (*) COUNT_MORE_SAL
FROM EMPLOYEES E1, EMPLOYEES E2
WHERE E2.SALARY > E1.SALARY
GROUP BY E1.EMPLOYEE_ID, E1.FIRST_NAME;   //  both are must in group by. first name must coz we print it. id must to handle duplicate names.



##
find how many people each one manages, zero included

SELECT M.FIRST_NAME, COUNT (E.EMPLOYEE_ID) TOTAL_MANAGED
FROM EMPLOYEES M
LEFT OUTER JOIN EMPLOYEES E
ON M.EMPLOYEE_ID = E.MANAGER_ID
GROUP_BY M.EMPLOYEE_ID, M.FIRST_NAME;

SELECT LAST_NAME, SALARY, MAX_SAL, MIN_SAL
FROM EMPLOYEES
JOIN (  -- create a custom table that has dept_id, max_sal, min_sal
    SELECT DEPARTMENT_ID, MAX(SALARY) MAX_SAL, MIN(SALARY) MIN_SAL
    FROM EMPLOYEES
    GROUP BY DEPARTMENT_ID ) MIN_MAX_TABLE
ON EMPLOYEES.DEPARTMENT_ID = MIN_MAX_TABLE.DEPARTMENT_ID



## example of sub-query inside WHERE
find last_name, salary of people having higher salary than everyone/anyone whose last name starts with K
SELECT LAST_NAME, SALARY
FROM EMPLOYEES
WHERE SALARY > ALL (SELECT SALARY FROM EMPLOYEES WHERE LAST_NAME LIKE 'K%')

for everyone, use ALL
for anyone, use ANY
