2.1.e 	[case, when, then, else, end]
SELECT LOCATION_ID, 
    (STREET_ADDRESS || ', ' || CITY || ', ' ||
        CASE WHEN STATE_PROVINCE IS NULL
             THEN ''
             ELSE (STATE_PROVINCE || ', ')
        END
    || POSTAL_CODE) AS ADDRESS
FROM LOCATIONS;

retrieves name and job id of all employees whose job id field contains the word ‘CLERK’ without using %
SELECT CONCAT(FIRST_NAME, LAST_NAME) NAME, JOB_ID 
FROM EMPLOYEES 
WHERE INSTR( UPPER(JOB_ID), 'CLERK') > 0 ;

***  Print hire dates of all employees in the format 13th February, 1998:  ***
SELECT 
  TO_CHAR(hire_date, 'DD') ||
  CASE
    WHEN TO_CHAR(hire_date, 'DD') IN ('11','12','13') THEN 'th'
    WHEN TO_CHAR(hire_date, 'DD') LIKE '%1' THEN 'st'
    WHEN TO_CHAR(hire_date, 'DD') LIKE '%2' THEN 'nd'
    WHEN TO_CHAR(hire_date, 'DD') LIKE '%3' THEN 'rd'
    ELSE 'th'
  END || ' ' ||
  TO_CHAR(hire_date, 'Month, YYYY') AS hire_date_formatted
FROM employees;



day : DD, Ddspth
month : MM, MON, Month
year : YY, YEAR
when converting date to char, use format string as you wish



example : find min, max salaries of departments excluding dept. no. 80
SELECT DEPARTMENT_ID, MAX(SALARY), MIN(SALARY) 
FROM EMPLOYEES  
GROUP BY DEPARTMENT_ID 
HAVING DEPARTMENT_ID <> 80 ;



*** Advanced group by queries: ***
finds the total employees hired in each year.  
SELECT TO_CHAR(HIRE_DATE, 'YYYY') YEAR, COUNT(*) TOTAL 
FROM EMPLOYEES  
GROUP BY TO_CHAR(HIRE_DATE, 'YYYY')  
ORDER BY YEAR ASC ;



*****  count for each character (except missing ones) the count of names starting with it: *****
SELECT SUBSTR(FIRST_NAME, 1, 1) FC, COUNT(*) TOTAL 
FROM EMPLOYEES  
GROUP BY SUBSTR(FIRST_NAME, 1, 1)  
ORDER BY FC ASC ;
